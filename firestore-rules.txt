rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Allow public read access to products, collections, and public general settings
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null || true; // Adjust to request.auth != null when Firebase Auth is enforced
    }

    match /collections/{collectionId} {
      allow read: if true;
      allow write: if request.auth != null || true;
    }

    match /settings/{settingId} {
      allow read: if true;
      allow write: if request.auth != null || true;
    }
  }
}
